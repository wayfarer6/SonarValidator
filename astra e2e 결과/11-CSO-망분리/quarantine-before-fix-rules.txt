table ip filter {
	chain input {
		type filter hook input priority filter; policy accept;
	}

	chain forward {
		type filter hook forward priority filter; policy accept;
	}

	chain output {
		type filter hook output priority filter; policy accept;
	}
}
table ip nat {
	chain postrouting {
		type nat hook postrouting priority srcnat; policy accept;
		oifname "eth0" masquerade
	}
}
table inet astra_cso {
	set confidential_v4 {
		type ipv4_addr
		flags interval
		elements = { 10.10.131.0/24, 10.10.132.0/24,
			     10.10.133.0/24 }
	}

	set sensitive_v4 {
		type ipv4_addr
		flags interval
		elements = { 10.20.111.0/24, 10.20.112.0/24,
			     10.40.121.0/24, 10.40.122.0/24 }
	}

	chain forward {
		type filter hook forward priority -50; policy accept;
		meta nfproto ipv6 iifname { "eth1.131", "eth1.132", "eth1.133" } counter packets 0 bytes 0 drop comment "CSO-no-IPv6-bypass-out"
		meta nfproto ipv6 oifname { "eth1.131", "eth1.132", "eth1.133" } counter packets 0 bytes 0 drop comment "CSO-no-IPv6-bypass-in"
		iifname "eth1.131" ip saddr != 10.10.131.0/24 counter packets 0 bytes 0 drop comment "CSO-anti-spoof-131"
		iifname "eth1.132" ip saddr != 10.10.132.0/24 counter packets 0 bytes 0 drop comment "CSO-anti-spoof-132"
		iifname "eth1.133" ip saddr != 10.10.133.0/24 counter packets 0 bytes 0 drop comment "CSO-anti-spoof-133"
		ip saddr @confidential_v4 ip daddr @confidential_v4 counter packets 0 bytes 0 accept comment "CSO-C-to-C"
		ip saddr @confidential_v4 ip daddr @sensitive_v4 counter packets 0 bytes 0 accept comment "CSO-C-to-S"
		ip saddr @sensitive_v4 ip daddr @confidential_v4 counter packets 0 bytes 0 accept comment "CSO-S-to-C"
		ip saddr @confidential_v4 counter packets 0 bytes 0 drop comment "CSO-C-to-Open-Internet-deny"
		ip daddr @confidential_v4 counter packets 0 bytes 0 drop comment "CSO-Open-Internet-to-C-deny"
	}
}
